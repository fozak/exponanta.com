Get-ChildItem -Recurse -Filter index.html | ForEach-Object {
    $file = $_.FullName
    $content = Get-Content -LiteralPath $file -Raw
    
    if ($content -match '<link rel="canonical" href="index\.html"') {
        if ($content -match '<meta property="og:url" content="([^"]+)"') {
            $url = $matches[1]
            $fixed = $content -replace '<link rel="canonical" href="index\.html" />', "<link rel=`"canonical`" href=`"$url`" />"
            Set-Content -LiteralPath $file -Value $fixed -NoNewline
            Write-Host "Fixed: $file → $url"
        } else {
            Write-Host "SKIPPED (no og:url): $file"
        }
    }
}
